cmake_minimum_required(VERSION 3.16)
project(libtree VERSION 0.2.0 LANGUAGES CXX)
find_package(Threads REQUIRED)
option(LIBTREE_SANITIZE "Enable address, undefined behavior and leak checks" OFF)
option(LIBTREE_COVERAGE "Instrument native code for gcov" OFF)
option(LIBTREE_THREAD_SANITIZE "Instrument native code for data races" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(LIBTREE_SANITIZE AND LIBTREE_COVERAGE)
  message(FATAL_ERROR "Use separate sanitizer and coverage builds")
endif()
if(LIBTREE_THREAD_SANITIZE AND (LIBTREE_SANITIZE OR LIBTREE_COVERAGE))
  message(FATAL_ERROR "Use a separate ThreadSanitizer build")
endif()
if(LIBTREE_THREAD_SANITIZE)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=thread)
endif()
if(LIBTREE_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=address,undefined)
endif()
if(LIBTREE_COVERAGE)
  # Concurrent tests need atomic gcov counters; ordinary counters lose updates.
  add_compile_options(--coverage -fprofile-update=atomic -O0 -g)
  add_link_options(--coverage)
endif()
add_library(tree src/gbdt.cc src/model.cc)
target_link_libraries(tree PUBLIC Threads::Threads)
set_target_properties(tree PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(tree PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
                                       $<INSTALL_INTERFACE:include>)
target_compile_options(tree PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_library(tree_c SHARED src/c_api.cc)
target_link_libraries(tree_c PRIVATE tree)
set_target_properties(tree_c PROPERTIES PREFIX "" OUTPUT_NAME "libtree")
target_compile_options(tree_c PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
foreach(test gbdt c_api parallel)
  add_executable(${test}_test tests/${test}_test.cc)
  target_link_libraries(${test}_test PRIVATE tree tree_c)
  target_compile_options(${test}_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME ${test} COMMAND ${test}_test)
endforeach()
add_executable(libtree_example examples/train.cc)
target_link_libraries(libtree_example PRIVATE tree)
add_test(NAME example COMMAND libtree_example)
add_executable(xgbt src/cli.cc)
target_link_libraries(xgbt PRIVATE tree)
target_compile_options(xgbt PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME cli_help COMMAND xgbt --help)
add_executable(native_benchmark benchmarks/native.cc)
target_link_libraries(native_benchmark PRIVATE tree ${CMAKE_DL_LIBS})
target_compile_options(native_benchmark PRIVATE -Wall -Wextra -Wpedantic -Werror)
install(TARGETS tree tree_c ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(TARGETS xgbt RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
